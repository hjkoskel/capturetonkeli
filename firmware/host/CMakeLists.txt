# Host build of the firmware core: unit tests + ctp_fw_host (core on a PTY).
cmake_minimum_required(VERSION 3.22)
project(ctp_host C)

set(CTP_C_STD "c11" CACHE STRING "C standard for the core: c11 | gnu23")
option(CTP_SANITIZE "Build with ASan/UBSan" ON)
option(CTP_COVERAGE "Build with gcov instrumentation" OFF)
option(CTP_ANALYZER "gcc -fanalyzer" OFF)

if(CTP_C_STD STREQUAL "gnu23")
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -std=gnu2x")
else()
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -std=c11")
endif()

set(CTP_WARN -Wall -Wextra -Wconversion -Wsign-conversion -Wshadow -Wformat=2 -Wstrict-prototypes
    -Wmissing-prototypes -Wswitch-enum -Wundef -Wnull-dereference -Werror)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    list(APPEND CTP_WARN -Wcast-align=strict)
endif()
if(CTP_ANALYZER)
    list(APPEND CTP_WARN -fanalyzer)
endif()
add_compile_options(-g -O1 -D_POSIX_C_SOURCE=200809L -D_DEFAULT_SOURCE)
if(CTP_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
if(CTP_COVERAGE)
    add_compile_options(--coverage)
    add_link_options(--coverage)
endif()

set(REPO ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_subdirectory(../components/ctp_core ctp_core)
target_compile_options(ctp_core PRIVATE ${CTP_WARN})

add_library(unity STATIC unity/unity.c)
target_include_directories(unity PUBLIC unity)

# unit tests (shared vectors from testvectors/c, generated by `go generate ./...`)
add_executable(ctp_tests tests/test_main.c tests/test_codec.c tests/test_filter.c tests/test_ring.c tests/test_proto.c)
target_include_directories(ctp_tests PRIVATE ${REPO}/testvectors/c ../components/ctp_core/src)
target_compile_options(ctp_tests PRIVATE ${CTP_WARN} -Wno-missing-prototypes)
target_link_libraries(ctp_tests ctp_core unity)

# firmware-in-the-loop: real core + host ports on a PTY
add_executable(ctp_fw_host fw_host_main.c port_host/usb_pty.c port_host/radio_fake.c port_host/sys_host.c)
target_include_directories(ctp_fw_host PRIVATE port_host ${REPO}/testvectors/c)
target_compile_options(ctp_fw_host PRIVATE ${CTP_WARN})
target_link_libraries(ctp_fw_host ctp_core pthread)

# fuzz targets: libFuzzer with clang (-DCMAKE_C_COMPILER=clang), otherwise a
# standalone random-input driver under the same sanitizers.
foreach(t frame filter messages)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang")
        add_executable(fuzz_${t} fuzz/fuzz_${t}.c fuzz/libfuzzer_assert.c)
        target_compile_options(fuzz_${t} PRIVATE -fsanitize=fuzzer)
        target_link_options(fuzz_${t} PRIVATE -fsanitize=fuzzer)
    else()
        add_executable(fuzz_${t} fuzz/fuzz_${t}.c fuzz/standalone.c)
    endif()
    target_compile_options(fuzz_${t} PRIVATE ${CTP_WARN})
    target_link_libraries(fuzz_${t} ctp_core)
endforeach()

# differential test helper (C filter vs Go filter.Match; internal/difftest)
add_executable(filter_diff tools/filter_diff.c)
target_compile_options(filter_diff PRIVATE ${CTP_WARN} -Wno-format-nonliteral)
target_link_libraries(filter_diff ctp_core)
